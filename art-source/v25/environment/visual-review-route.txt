'use client';
// Temporary visual QA route, removed before delivery.
import {useEffect,useRef,useState} from 'react';
import {GameEngine} from '@/lib/game/engine';
import {WorldRenderer} from '@/lib/game/renderer';
import {MAPS,type MapId} from '@/lib/game/data';
import {ENVIRONMENT_THEMES} from '@/lib/game/environment';

export default function EnvironmentReview(){
 const canvas=useRef<HTMLCanvasElement>(null),renderer=useRef<WorldRenderer|null>(null),[mapId,setMapId]=useState<MapId>('patio'),[ready,setReady]=useState(false),[overview,setOverview]=useState(true),[avaDirection,setAvaDirection]=useState<number|null>(null);
 useEffect(()=>{const game=new GameEngine();game.start(false,'seiji');game.finishCutscene();game.state.stage=5;game.state.mode='world';const world=new WorldRenderer(canvas.current!,game);renderer.current=world;world.load().then(()=>{world.destroy();setReady(true);});return()=>world.destroy();},[]);
 useEffect(()=>{if(!ready||!renderer.current||!canvas.current)return;const world=renderer.current,map=MAPS[mapId],ctx=world.ctx,cv=canvas.current;world.engine.state.map=mapId;world.engine.state.position={x:map.width/2,y:map.height/2};world.frame=300;
  if(avaDirection!==null){cv.width=1260;cv.height=500;cv.style.width='100%';cv.style.height='auto';ctx.setTransform(1,0,0,1,0,0);ctx.fillStyle='#15253a';ctx.fillRect(0,0,1260,500);for(let i=0;i<5;i++){const x=126+i*250;ctx.strokeStyle='#5b6d72';ctx.beginPath();ctx.moveTo(x,30);ctx.lineTo(x,435);ctx.stroke();ctx.save();ctx.translate(x,410);ctx.scale(2.9,2.9);world.frame=Math.max(0,i-1)*155+1;world.actor('ava',0,0,avaDirection,i>0);ctx.restore();ctx.font='18px Arial';ctx.fillStyle='#eddfc3';ctx.textAlign='center';ctx.fillText(i===0?'Parada':`Passo ${i}`,x,472);}return;}
  if(!overview){world.disposed=false;world.draw();return()=>{world.disposed=true;cancelAnimationFrame(world.raf);};}
  const width=1260,height=800,scale=Math.min((width-80)/(map.width*56),(height-50)/(map.height*56));cv.width=width;cv.height=height;cv.style.width='100%';cv.style.height='auto';ctx.setTransform(1,0,0,1,0,0);ctx.fillStyle='#081624';ctx.fillRect(0,0,width,height);ctx.translate((width-map.width*56*scale)/2+28*scale,(height-map.height*56*scale)/2+28*scale);ctx.scale(scale,scale);ctx.drawImage(world.ground(map),-28,-28);
  const things=map.props.map((prop,index)=>({y:prop.y,draw:()=>world.scenery(map,prop,index)}));things.push(...map.entities.map(entity=>({y:entity.y,draw:()=>world.entity(entity)})));things.sort((a,b)=>a.y-b.y).forEach(thing=>thing.draw());ctx.setTransform(1,0,0,1,0,0);const shade=ctx.createLinearGradient(0,0,0,height);shade.addColorStop(0,ENVIRONMENT_THEMES[mapId].atmosphere[0]);shade.addColorStop(1,ENVIRONMENT_THEMES[mapId].atmosphere[1]);ctx.fillStyle=shade;ctx.fillRect(0,0,width,height);
 },[mapId,ready,overview,avaDirection]);
 return <main style={{padding:16,background:'#0b1723',color:'#f1e5c8',minHeight:'100vh',fontFamily:'Arial'}}><h1 style={{fontSize:22}}>Revisão visual de cenários v25</h1><p>Prévia de auditoria: {MAPS[mapId].name} · {ready?'artes carregadas':'carregando artes'}</p><nav style={{display:'flex',gap:8,flexWrap:'wrap',marginBottom:12}}>{Object.values(MAPS).map(map=><button key={map.id} onClick={()=>{setMapId(map.id);setAvaDirection(null);}} style={{background:map.id===mapId?'#dcc28b':'#233449',color:map.id===mapId?'#182333':'#e7ddc9',padding:'8px 12px',border:'1px solid #9d8b69',borderRadius:8}}>{map.name}</button>)}</nav><div style={{display:'flex',gap:8,marginBottom:8}}><button onClick={()=>{setOverview(!overview);setAvaDirection(null);}} style={{padding:8}}>{overview?'Ver câmera normal':'Ver cenário inteiro'}</button>{['Sul','Oeste','Leste','Norte'].map((direction,index)=><button key={direction} onClick={()=>setAvaDirection(index)} style={{padding:8}}>Ava · {direction}</button>)}</div><canvas ref={canvas} style={{width:'100%',height:650,border:'1px solid #847856',display:'block',maxWidth:avaDirection===null&&overview?740:1160}} aria-label={`Cenário ${MAPS[mapId].name}`}/></main>;
}
